// public/js/players.js: STU-UX2 audio cards: a waveform player (play/pause, click to seek, time) drawn from the
// file itself. Cards are re-rendered often, so everything works by delegation and new players are drawn as they appear.
(function () {
  'use strict';

  const AudioCtx = window.AudioContext || window.webkitAudioContext;
  const peaksCache = new Map();

  // The loudest sample in each of 200 slices of channel 0 (cached per URL)
  function peaks(url) {
    if (!peaksCache.has(url)) {
      peaksCache.set(url, (async () => {
        const response = await fetch(url);
        if (!response.ok) throw new Error(`HTTP ${response.status}`);
        const audioCtx = new AudioCtx();
        try {
          const audioBuffer = await audioCtx.decodeAudioData(await response.arrayBuffer());
          const channelData = audioBuffer.getChannelData(0);
          const step = Math.max(1, Math.floor(channelData.length / 200));
          const peakValues = [];
          for (let i = 0; i < 200; i++) {
            let maxVal = 0;
            for (let j = i * step; j < (i + 1) * step && j < channelData.length; j++) {
              const val = Math.abs(channelData[j]);
              if (val > maxVal) maxVal = val;
            }
            peakValues.push(maxVal);
          }
          return peakValues;
        } finally {
          audioCtx.close();
        }
      })());
      peaksCache.get(url).catch(() => peaksCache.delete(url));
    }
    return peaksCache.get(url);
  }

  async function draw(player) {
    const canvas = player.querySelector('canvas.wave');
    const audio = player.querySelector('audio');
    if (!canvas || !audio || !player.dataset.src) return;

    let peakData;
    try {
      peakData = await peaks(player.dataset.src);
    } catch {
      return; // undrawn, no error on the page
    }

    const dpr = window.devicePixelRatio || 1;
    canvas.width = (canvas.clientWidth || 300) * dpr;
    canvas.height = (canvas.clientHeight || 48) * dpr;
    const top = Math.max(...peakData) || 1;
    const progress = audio.duration ? audio.currentTime / audio.duration : 0;
    const barWidth = canvas.width / peakData.length;

    const ctx = canvas.getContext('2d');
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    ctx.fillStyle = getComputedStyle(canvas).color;
    peakData.forEach((val, i) => {
      const h = Math.max(1, (val / top) * canvas.height);
      ctx.globalAlpha = i < peakData.length * progress ? 1 : 0.45;
      ctx.fillRect(i * barWidth, (canvas.height - h) / 2, Math.max(1, barWidth - dpr), h);
    });
    ctx.globalAlpha = 1;
    canvas.dataset.ready = '1';
  }

  function drawNew() {
    for (const canvas of document.querySelectorAll('.wave-player canvas.wave:not([data-ready])')) {
      draw(canvas.closest('.wave-player'));
    }
  }

  function timeText(seconds) {
    const s = Math.floor(seconds || 0);
    return `${Math.floor(s / 60)}:${String(s % 60).padStart(2, '0')}`;
  }

  function seek(audio, fraction) {
    audio.currentTime = fraction * audio.duration;
  }

  document.addEventListener('click', (e) => {
    const player = e.target.closest('.wave-player');
    if (!player) return;
    const audio = player.querySelector('audio');
    if (!audio) return;

    if (e.target.closest('.wave-play')) {
      if (audio.paused) audio.play().catch(() => {});
      else audio.pause();
    } else if (e.target.closest('canvas.wave')) {
      const canvas = e.target.closest('canvas.wave');
      const fraction = Math.min(1, Math.max(0, e.offsetX / (canvas.clientWidth || 1)));
      if (isNaN(audio.duration)) {
        audio.addEventListener('loadedmetadata', () => seek(audio, fraction), { once: true });
        audio.load();
      } else {
        seek(audio, fraction);
      }
    }
  });

  // Media events do not bubble: listen in the capture phase
  for (const type of ['play', 'pause', 'ended', 'timeupdate', 'seeked']) {
    document.addEventListener(type, (e) => {
      const player = e.target.closest && e.target.closest('.wave-player');
      if (!player) return;
      const button = player.querySelector('.wave-play');
      if (type === 'play') {
        button?.setAttribute('aria-label', 'Pause');
        player.classList.add('playing');
      } else if (type === 'pause' || type === 'ended') {
        button?.setAttribute('aria-label', 'Play');
        player.classList.remove('playing');
      } else {
        const time = player.querySelector('.wave-time');
        if (time) time.textContent = timeText(e.target.currentTime);
        draw(player);
      }
    }, true);
  }

  new MutationObserver(drawNew).observe(document.body, { childList: true, subtree: true });
  drawNew();
})();
