(function () {
  'use strict';
  const NS = typeof module === 'object' && module.exports ? null : (globalThis.KKPose = globalThis.KKPose || {});
  const need = (name, file) => (NS ? NS[name] : require(file));

  // Load vec.js inside the function to ensure correct loading order in browser
  const V = need('vec', './vec.js');

  // --- Data definitions ---

  const BODY = [
    ['pelvis', null, [0, 0, 0.95], 'spine', [0, -1, 0], [-180, 180, -180, 180, -180, 180]],
    ['spine', 'pelvis', [0, 0, 1.13], 'chest', [0, -1, 0], [-30, 45, -30, 30, -40, 40]],
    ['chest', 'spine', [0, 0, 1.36], 'neck', [0, -1, 0], [-20, 30, -20, 20, -30, 30]],
    ['neck', 'chest', [0, 0, 1.49], 'head', [0, -1, 0], [-40, 50, -30, 30, -60, 60]],
    ['head', 'neck', [0, 0, 1.68], 'head_top', [0, -1, 0], [-40, 40, -30, 30, -60, 60]],
    ['head_top', 'head', [0, 0, 1.80], null, [0, -1, 0], null],
    ['nose', 'head', [0, -0.105, 1.66], null, [0, -1, 0], null],
    ['l_eye', 'head', [0.035, -0.092, 1.70], null, [0, -1, 0], null],
    ['l_ear', 'head', [0.088, 0, 1.68], null, [0, -1, 0], null],
    ['l_shoulder', 'chest', [0.19, 0, 1.46], 'l_elbow', [0, -1, 0], [-60, 180, -40, 170, -90, 90]],
    ['l_elbow', 'l_shoulder', [0.29, 0, 1.18], 'l_wrist', [0, -1, 0], [0, 150, -10, 10, -90, 90]],
    ['l_wrist', 'l_elbow', [0.37, -0.03, 0.93], 'l_middle_mcp', 'PALM', [-70, 80, -30, 40, -20, 20]],
    ['l_hip', 'pelvis', [0.10, 0, 0.95], 'l_knee', [0, -1, 0], [-30, 130, -30, 70, -45, 45]],
    ['l_knee', 'l_hip', [0.11, -0.01, 0.52], 'l_ankle', [0, -1, 0], [-150, 0, -5, 5, -30, 30]],
    ['l_ankle', 'l_knee', [0.11, 0.01, 0.09], 'l_toe', [0, 0, -1], [-30, 50, -25, 25, -20, 20]],
    ['l_toe', 'l_ankle', [0.11, -0.13, 0.02], null, [0, -1, 0], null],
  ];

  const FINGER_UVW = {
    thumb: [[0.025, 0.020, 0.010], [0.050, 0.035, 0.015], [0.075, 0.045, 0.020], [0.095, 0.050, 0.020]],
    index: [[0.090, 0.025, 0], [0.130, 0.027, 0], [0.155, 0.028, 0], [0.175, 0.029, 0]],
    middle: [[0.095, 0, 0], [0.140, 0, 0], [0.168, 0, 0], [0.190, 0, 0]],
    ring: [[0.090, -0.010, 0], [0.130, -0.011, 0], [0.155, -0.012, 0], [0.175, -0.013, 0]],
    pinky: [[0.080, -0.027, 0], [0.110, -0.030, 0], [0.130, -0.032, 0], [0.146, -0.034, 0]],
  };

  const FINGER_LIMITS = {
    thumb: [[-20, 50, -30, 40, -10, 10], [0, 60, 0, 0, 0, 0], [-10, 80, 0, 0, 0, 0], null],
    others: [[-20, 90, -20, 20, -5, 5], [0, 110, 0, 0, 0, 0], [0, 80, 0, 0, 0, 0], null],
  };

  const FINGERS = {
    thumb: ['thumb_cmc', 'thumb_mcp', 'thumb_ip'],
    index: ['index_mcp', 'index_pip', 'index_dip'],
    middle: ['middle_mcp', 'middle_pip', 'middle_dip'],
    ring: ['ring_mcp', 'ring_pip', 'ring_dip'],
    pinky: ['pinky_mcp', 'pinky_pip', 'pinky_dip'],
  };

  const HAND = ['wrist', 'thumb_cmc', 'thumb_mcp', 'thumb_ip', 'thumb_tip', 'index_mcp', 'index_pip', 'index_dip', 'index_tip', 'middle_mcp', 'middle_pip', 'middle_dip', 'middle_tip', 'ring_mcp', 'ring_pip', 'ring_dip', 'ring_tip', 'pinky_mcp', 'pinky_pip', 'pinky_dip', 'pinky_tip'];

  const mirrorName = (name) => {
    if (name.startsWith('l_')) return name.replace('l_', 'r_');
    if (name.startsWith('r_')) return name.replace('r_', 'l_');
    return name;
  };

  // --- Build JOINTS ---

  const JOINTS = [];
  const LEAF = { bend: [0, 0], side: [0, 0], twist: [0, 0] };
  const limitsOf = (l) => (l ? { bend: [l[0], l[1]], side: [l[2], l[3]], twist: [l[4], l[5]] } : LEAF);
  const mirror = (v) => [-v[0], v[1], v[2]];
  const restOf = {};
  for (const row of BODY) restOf[row[0]] = row[2];

  // 1. Palm normal n and thumb axis t of each hand (the right hand from the mirrored left rest)
  const axes = {};
  for (const s of ['l', 'r']) {
    const pick = (name) => (s === 'l' ? restOf[name] : mirror(restOf[name]));
    const wrist = pick('l_wrist');
    const d = V.norm(V.sub(wrist, pick('l_elbow')));
    const n0 = s === 'l' ? [-1, 0, 0] : [1, 0, 0];
    const n = V.norm(V.sub(n0, V.scale(d, V.dot(n0, d))));
    const t = s === 'l' ? V.cross(n, d) : V.cross(d, n);
    axes[s] = { wrist, d, n, t };
  }

  // 2. Body rows, each l_ row followed by its r_ twin
  for (const [name, parent, rest, aim, front, limits] of BODY) {
    JOINTS.push({ name, parent, rest, aim, front: front === 'PALM' ? axes.l.n : front, limits: limitsOf(limits) });
    if (!name.startsWith('l_')) continue;
    JOINTS.push({
      name: mirrorName(name),
      parent: mirrorName(parent),
      rest: mirror(rest),
      aim: aim && mirrorName(aim),
      front: front === 'PALM' ? axes.r.n : mirror(front),
      limits: limitsOf(limits),
    });
  }

  // 3. Hands: per finger 4 points, wrist + u*d + v*t + w*n
  for (const s of ['l', 'r']) {
    const { wrist, d, n, t } = axes[s];
    for (const f of Object.keys(FINGERS)) {
      const names = [...FINGERS[f], f + '_tip'];
      const lim = FINGER_LIMITS[f === 'thumb' ? 'thumb' : 'others'];
      for (let k = 0; k < 4; k++) {
        const [u, v, w] = FINGER_UVW[f][k];
        JOINTS.push({
          name: `${s}_${names[k]}`,
          parent: k === 0 ? `${s}_wrist` : `${s}_${names[k - 1]}`,
          rest: V.add(V.add(V.add(wrist, V.scale(d, u)), V.scale(t, v)), V.scale(n, w)),
          aim: k < 3 ? `${s}_${names[k + 1]}` : null,
          front: n,
          limits: limitsOf(lim[k]),
        });
      }
    }
  }

  // --- Export ---

  const byName = {};
  for (const j of JOINTS) {
    byName[j.name] = j;
  }

  const api = {
    JOINTS,
    byName,
    HAND,
    BODY18: ['nose', 'neck', 'r_shoulder', 'r_elbow', 'r_wrist', 'l_shoulder', 'l_elbow', 'l_wrist', 'r_hip', 'r_knee', 'r_ankle', 'l_hip', 'l_knee', 'l_ankle', 'r_eye', 'l_eye', 'r_ear', 'l_ear'],
    FINGERS,
    mirrorName
  };

  if (NS) NS.skeleton = api; else module.exports = api;
})();
