//! Runner self-update (RUNNER-SYNC): check a downloaded binary and swap it in.

use std::path::Path;

/// True when `sha_text` (the content of a `.sha256` file: a hex digest, optionally followed by a file name) is the SHA-256 of `bytes`.
pub fn sha_ok(bytes: &[u8], sha_text: &str) -> bool {
    use sha2::{Digest, Sha256};
    let expected = Sha256::digest(bytes).iter().map(|b| format!("{b:02x}")).collect::<String>().to_lowercase();
    let first_word = sha_text.split_whitespace().next()?;
    first_word.eq_ignore_ascii_case(&expected)
}

/// Writes `bytes` next to `target` as `<target>.new`, makes it executable (0o755), runs it with `--version`
/// and requires its output to start with `kompanion-runner `; only then renames it over `target`.
/// On any failure the `.new` file is removed, `target` is untouched and the error text is returned.
pub fn install_binary(bytes: &[u8], target: &Path) -> Result<(), String> {
    use std::fs::{self, File};
    use std::io::Write;
    use std::os::unix::fs::{PermissionsExt, Permissions};
    use std::process::Command;

    let new_path = target.with_extension("new");
    
    // Write bytes to .new file
    let mut file = File::create(&new_path).map_err(|e| e.to_string())?;
    file.write_all(bytes).map_err(|e| e.to_string())?;
    
    // Make executable (0o755)
    let metadata = fs::metadata(&new_path).map_err(|e| e.to_string())?;
    let mut perms = metadata.permissions();
    perms.set_mode(0o755);
    fs::set_permissions(&new_path, perms).map_err(|e| e.to_string())?;
    
    // Run with --version
    let output = Command::new(&new_path)
        .arg("--version")
        .output()
        .map_err(|e| e.to_string())?;
    
    if !output.status.success() {
        fs::remove_file(&new_path).ok();
        return Err(format!("Binary exited with status {}", output.status));
    }
    
    let stdout = String::from_utf8_lossy(&output.stdout);
    if !stdout.starts_with("kompanion-runner ") {
        fs::remove_file(&new_path).ok();
        return Err(format!("Output did not start with 'kompanion-runner ': {}", stdout.trim()));
    }
    
    // Rename over target
    fs::rename(&new_path, target).map_err(|e| e.to_string())?;
    
    Ok(())
}

#[cfg(test)]
mod test_install_binary {
    use super::*;
    use std::env;
    use std::fs;
    use tempfile::tempdir;

    #[test]
    fn test_install_binary_success() {
        let dir = tempdir().unwrap();
        let target = dir.path().join("runner");
        let bytes = b"kompanion-runner 1.0.0\n";
        
        assert!(install_binary(bytes, &target).is_ok());
        assert!(target.exists());
        assert!(!target.with_extension("new").exists());
    }

    #[test]
    fn test_install_binary_bad_version() {
        let dir = tempdir().unwrap();
        let target = dir.path().join("runner");
        let bytes = b"wrong-binary 1.0.0\n";
        
        let result = install_binary(bytes, &target);
        assert!(result.is_err());
        assert!(!target.exists());
        assert!(!target.with_extension("new").exists());
    }

    #[test]
    fn test_install_binary_write_fail() {
        let dir = tempdir().unwrap();
        let target = dir.path().join("runner");
        let bytes = b"kompanion-runner 1.0.0\n";
        
        // Make target read-only to simulate write failure
        let mut perms = fs::metadata(&target).unwrap().permissions();
        perms.set_readonly(true);
        fs::set_permissions(&target, perms).unwrap();
        
        let result = install_binary(bytes, &target);
        assert!(result.is_err());
        assert!(!target.with_extension("new").exists());
    }
}
