"""GATE-IN-CI: post the review gate result of every open PR as the commit status review-gate / gate."""

import os
import json
import urllib.request

HERE = os.path.dirname(os.path.abspath(__file__))
CONTEXT = "review-gate / gate"


def status_for(code):
    if code == 0:
        return ("success", "review findings are all handled")
    if code == 2:
        return ("pending", "PR-Agent is still reviewing the head")
    if code == 1:
        return ("failure", "review findings are open or unchecked")
    return ("error", "the review gate could not run (exit %d)" % code)


class Api:
    def __init__(self, pulls):
        self.pulls = pulls
        self.posted = []

    def open_pulls(self):
        return self.pulls

    def post_status(self, sha, state, description):
        self.posted.append((sha, state, description))


def sweep(api, gate_fn):
    out = {}
    for pr in api.open_pulls():
        if pr.get("draft"):
            continue
        try:
            code = gate_fn(pr["number"])
            state, desc = status_for(code)
            api.post_status(pr["head"]["sha"], state, desc)
            out[pr["number"]] = state
        except Exception:
            sha = pr["head"]["sha"]
            api.post_status(sha, "error", "the review gate crashed")
            out[pr["number"]] = "error"
    return out


class StatusClient:
    def __init__(self, url, owner, repo, token):
        self.url = url
        self.owner = owner
        self.repo = repo
        self.token = token

    def post_status(self, sha, state, description):
        req = urllib.request.Request(
            f"{self.url}/repos/{self.owner}/{self.repo}/statuses/{sha}",
            data=json.dumps({"state": state, "context": CONTEXT, "description": description}).encode(),
            headers={"Authorization": f"token {self.token}", "Content-Type": "application/json"},
            method="POST",
        )
        with urllib.request.urlopen(req) as resp:
            return resp.status
