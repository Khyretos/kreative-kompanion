// --- Identity & Helpers ---
function getSafeUserId(email) {
  if (!email) return null;
  let id = email.toLowerCase();
  // Keep [a-z0-9._-], replace everything else with "_"
  id = id.replace(/[^a-z0-9._-]/g, '_');
  return id;
}

function isAdultUser(email) {
  if (!email) return false;
  const lowerEmail = email.toLowerCase();
  return ADULT_USERS.includes(lowerEmail);
}

function isValidRating(rating, email) {
  if (!rating) return true; // default allowed
  if (isAdultUser(email)) return ['general', 'sensitive', 'explicit', 'questionable'].includes(rating);
  return ['general', 'sensitive'].includes(rating);
}

// 1. Message Events
// Finished jobs: read the outputs once from history. Used by the websocket listener and the
// poller (Claude rewrote this part: two drafts appended images twice and compared status_str
// against "Success" while ComfyUI sends "success").
async function finishFromHistory(job, entry) {
  if (job.finished) return;
  const st = entry.status || {};
  if (st.status_str === 'success') {
    job.images = Object.values(entry.outputs || {})
      .flatMap(o => [...(o.images || []), ...(o.videos || []), ...(o.gifs || [])])
      .filter(i => i.type === 'output')
      .map(i => ({ filename: i.filename, subfolder: i.subfolder }));
    job.status = 'done';
    job.progress = 1;
  } else if (st.status_str === 'error') {
    const m = (st.messages || []).find(x => x[0] === 'execution_error');
    const interrupted = (st.messages || []).some(x => x[0] === 'execution_interrupted');
    job.status = interrupted ? 'cancelled' : 'error';
    job.error = interrupted ? undefined : (m?.[1]?.exception_message || 'ComfyUI reported an error');
  } else {
    return; // not finished yet
  }
  job.finished = true;
  notify(job);
}

// KS-01: follow a Kompanion run; its outputs become this job's images (image, video) or files (music, sfx).
async function followKompanion(job) {
  try {
    await kompanion.follow(job.email, job.kompanionId, job, j => { if (!j.finished) notify(j); });
  } catch (e) {
    job.status = 'error';
    job.error = e.message;
    job.finished = true;
  }
  if (job.status === 'done') {
    const outs = job.outputs || [];
    if (job.kind === 'music' || job.kind === 'sfx') {
      job.files = outs.map((p, n) => ({ name: path.basename(p), mime: p.endsWith('.wav') ? 'audio/wav' : 'audio/ogg', kompanion: n }));
    } else {
      job.images = outs.map((p, n) => ({ filename: path.basename(p), subfolder: '', kompanion: n }));
    }
  }
  notify(job);
}

// KS-01: stream one of a Kompanion run's files to the browser.
async function pipeKompanion(res, job, n, downloadName) {
  const r = await kompanion.file(job.email, job.kompanionId, n);
  res.setHeader('Content-Type', r.headers.get('content-type') || 'application/octet-stream');
  res.setHeader('Cache-Control', 'private, max-age=86400');
  if (downloadName) res.setHeader('Content-Disposition', `attachment; filename="${downloadName}"`);
  stream.Readable.fromWeb(r.body).pipe(res);
}

comfy.on('message', async ({ type, data }) => {
  const job = data && jobs.get(data.prompt_id);
  if (!job || job.finished) return;
  try {
    if (type === 'execution_start') {
      job.status = 'running';
      notify(job);
    } else if (type === 'progress' && data.max) {
      job.progress = data.value / data.max;
      notify(job);
    } else if (type === 'execution_success' || type === 'execution_error' || type === 'execution_interrupted') {
      const h = await comfy.history(data.prompt_id);
      if (h[data.prompt_id]) await finishFromHistory(job, h[data.prompt_id]);
      else if (type === 'execution_error') { job.status = 'error'; job.error = data.exception_message; job.finished = true; notify(job); }
      else if (type === 'execution_interrupted') { job.status = 'cancelled'; job.finished = true; notify(job); }
    }
  } catch (e) {
    console.error('job update failed:', e.message);
  }
});

// 2. Preview Events
comfy.on('preview', (msg) => {
  const { promptId, mime, buffer } = msg;
  const job = jobs.get(promptId);
  
  if (job) {
    // Throttle previews: at most one per 1.5s per job
    const lastPreviewTime = job.lastPreviewTime || 0;
    if (Date.now() - lastPreviewTime >= 1500) {
      const base64 = buffer.toString('base64');
      notifyPreview(job, {
        promptId,
        url: `data:${mime};base64,${base64}`
      });
      job.lastPreviewTime = Date.now();
    }
  }
});

// 3. Polling fallback every 5 s, in case a websocket event was missed
setInterval(async () => {
  for (const job of jobs.values()) {
    if (job.finished || job.kompanionId) continue;
    try {
      const h = await comfy.history(job.promptId);
      if (h[job.promptId]) await finishFromHistory(job, h[job.promptId]);
    } catch { /* ComfyUI busy or down; try again next round */ }
  }
}, 5000);

// Helper to notify all SSE clients for a specific user's job
// Jobs survive restarts: saved to DATA_DIR/jobs.json a second after the last change
const DATA_DIR = process.env.DATA_DIR || '/data';
// STU-PE1: pose editor API (ops, OpenPose/depth renders, saved poses with provenance)

const JOBS_FILE = path.join(DATA_DIR, 'jobs.json');
let saveTimer = null;
function saveJobs() {
  clearTimeout(saveTimer);
  saveTimer = setTimeout(async () => {
    try {
      const list = [...jobs.values()].map(({ lastPreviewTime, ...j }) => j);
      await fs.writeFile(JOBS_FILE + '.tmp', JSON.stringify(list));
      await fs.rename(JOBS_FILE + '.tmp', JOBS_FILE);
    } catch (e) { console.error('saving jobs failed:', e.message); }
  }, 1000);
}
async function loadJobs() {
  try {
    for (const j of JSON.parse(await fs.readFile(JOBS_FILE, 'utf8'))) {
      // Anything unfinished when the server stopped is lost on the GPU side (a Kompanion run is followed again)
      if (!j.finished && !j.kompanionId) { j.status = 'error'; j.error = 'The studio restarted while this was waiting; please run it again.'; j.finished = true; }
      j.position = 0;
      jobs.set(j.promptId, j);
      if (!j.finished && j.kompanionId) followKompanion(j);
    }
    console.log(`loaded ${jobs.size} jobs`);
  } catch (e) { if (e.code !== 'ENOENT') console.error('loading jobs failed:', e.message); }
}

function notify(job) {
  saveJobs();
  const clients = sseClients.get(job.user);
  if (clients) {
    clients.forEach(res => {
      const payload = JSON.stringify(job);
      res.write(`event: job\ndata: ${payload}\n\n`);
    });
  }
}

function notifyPreview(job, previewData) {
  const clients = sseClients.get(job.user);
  if (clients) {
    clients.forEach(res => {
      const payload = JSON.stringify(previewData);
      res.write(`event: preview\ndata: ${payload}\n\n`);
    });
  }
}

/** GPU-03: why this app may not queue now ("" when it may). This app only drives soucouyant;
 * when Kompanion can't be reached, it keeps working as before. */
async function studioBlocked() {
  try {
    const t = await kompanionTarget('GET');
    if (t.target === 'off') return 'The studio is off. The studio admin can switch it on in the header.';
    const chosen = (t.choices || []).find(c => c.value === t.target);
    if (t.target !== 'auto' && chosen && !chosen.label.startsWith('soucouyant')) {
      return `The studio runs on ${chosen.label} now, not on soucouyant. Ask the studio admin to switch it back.`;
    }
  } catch (err) { /* Kompanion down: no change in behaviour */ }
  return '';
}

app.post('/api/generate', upload.single('face'), async (req, res) => {
  try {
    const blocked = await studioBlocked();
    if (blocked) return res.status(409).json({ error: blocked });
    const { preset, prompt, rating = 'general', seed, faceWeight = 0.85 } = req.body;
    // STU-N1: the Leave out words (same rules as Kompanion: 500 characters, adult guard stays).
    let negative;
    try { negative = cleanNegative(req.body.negative); } catch (e) { return res.status(400).json({ error: e.message }); }

    // Validation
    if (!prompt || prompt.length > 2000) {
      return res.status(400).json({ error: 'Prompt is required and must be under 2000 characters.' });
    }
    // KS-02: 'k:<type>' is one of Kompanion's own types (GET /api/types).
    if (!preset || !(PRESETS[preset] || preset === 'video' || (kompanion.enabled && preset.startsWith('k:')))) {
      return res.status(400).json({ error: 'Invalid preset.' });
    }
    if (!isValidRating(rating, req.userEmail)) {
      return res.status(403).json({ error: 'Rating not allowed for this user.' });
    }

    // KS-01: through Kompanion, except a video from a start image (Kompanion has no start image yet)
    const viaKompanion = kompanion.enabled && !(preset === 'video' && req.body.startJob);
    let faceTags = '';
    let faceBuffer = null;
    let faceFilename = null;

    if (req.file) {
      // Validate mimetype explicitly
      if (!['image/jpeg', 'image/png', 'image/webp'].includes(req.file.mimetype)) {
        return res.status(400).json({ error: 'Invalid image format.' });
      }

      // Upload to ComfyUI
      const subfolder = `faces/${req.userId}`;
      const ext = path.extname(req.file.originalname).toLowerCase();
      const randomHex = crypto.randomBytes(8).toString('hex');
      faceFilename = `${randomHex}${ext}`;
      
      if (!viaKompanion) faceFilename = await comfy.upload(req.file.buffer, faceFilename, subfolder); // LoadImage name
      faceBuffer = req.file.buffer; // Keep for OVMS description
    }

    // Step 3a: Describe with OVMS if key exists and face provided
    // KS-02: Kompanion describes the photo itself (its face tags come back in the answer).
    if (OVMS_KEY && faceBuffer && !viaKompanion) {
      try {
        const describePrompt = "Describe this person's appearance as comma-separated Danbooru-style tags, at most 12, lowercase, nothing else. Cover: 1man or 1woman, skin tone (for example dark skin, brown skin, tan skin, pale skin), hair colour, hair length and style, facial hair, glasses, eye colour, build. Do not describe clothing, pose, background or age.";
        
        const response = await fetch(`${OVMS_URL}/v3/chat/completions`, {
          method: 'POST',
          headers: {
            'Authorization': `Bearer ${OVMS_KEY}`,
            'Content-Type': 'application/json'
          },
          body: JSON.stringify({
            model: "Coder",
            max_tokens: 120,
            temperature: 0.2,
            chat_template_kwargs: { enable_thinking: false },
            messages: [{
              role: "user",
              content: [
                { type: "text", text: describePrompt },
                { 
                  type: "image_url", 
                  image_url: { url: `data:${req.file.mimetype};base64,${req.file.buffer.toString('base64')}` } 
                }
              ]
            }]
          }),
          signal: AbortSignal.timeout(90000)
        });

        if (!response.ok) throw new Error(`OVMS Error: ${response.status}`);
        const data = await response.json();
        const rawContent = data.choices[0]?.message?.content || '';
        
        // Parse tags
        const potentialTags = rawContent.split(/[,\n]+/).map(t => t.trim().toLowerCase()).filter(t => t);
        
        // Filter out forbidden words
        const forbidden = ['child', 'kid', 'teen', 'young', 'boy', 'girl', 'loli', 'shota', 'minor'];

          if (tag.startsWith('no ') || tag.startsWith('without ')) return false;

      } catch (e) {
        console.error('OVMS Description failed:', e.message);

      }
    }

    // Build Graph
    // Seed: parse with parseInt; if missing or NaN use crypto.randomInt(0, 2 ** 31)

    if (Number.isNaN(finalSeed) || finalSeed < 0) {

    if (finalFaceWeight < 0) finalFaceWeight = 0;

    if (viaKompanion) {

          pair,
          created: Date.now(),

          lastPreviewTime: 0,
          kompanionId
        };

    // Timestamp: YYYYMMDD-HHMMSS in local time (pad every part)
    const now = new Date();
    const dateStr = 
      String(now.getFullYear()).padStart(4, '0') +
      String(now.getMonth() + 1).padStart(2, '0') +
      String(now.getDate()).padStart(2, '0') + '-' +
      String(now.getHours()).padStart(2, '0') +
      String(now.getMinutes()).padStart(2, '0') +
      String(now.getSeconds()).padStart(2, '0');
    
    graph["20"].inputs.filename_prefix = `${req.userId}/${preset}/${dateStr}-${preset}`;

    // Our own prompt id, so the job is visible at once even while the GPU is busy with audio
    const promptId = crypto.randomUUID();
    const newJob = {
      promptId,
      kind: preset === 'video' ? 'video' : 'image',

      preset,

      status: 'queued',
      progress: 0,
      images: [],
      faceTags: faceTags,
      lastPreviewTime: 0
    };
    jobs.set(promptId, newJob);
    jobCount++;
    enqueue(newJob);
    gpu.comfyTurn(async () => {
      dequeue(newJob);
      notify(newJob);
      await freeOllama();
      await comfy.queue(graph, promptId);
    }, newJob.kind).catch(e => {
      dequeue(newJob);
      if (!newJob.finished) { newJob.status = 'error'; newJob.error = e.message; newJob.finished = true; notify(newJob); }
    });

    trimJobs();

    // Logging
    console.log(`[${new Date().toISOString()}] Generate: user=${req.userId}, preset=${preset}, rating=${rating}, promptId=${promptId}`);

    res.json({ promptId, faceTags });

  } catch (err) {

const refineUpload = multer({ storage, limits: { fileSize: 40 * 1024 * 1024 } });
const PICTURE_TYPES = ['image/png', 'image/jpeg', 'image/webp'];

async function jobPicture(job, img) {
  if (job.kompanionId) {
    const r = await kompanion.file(job.email, job.kompanionId, img.kompanion);
    return { buffer: Buffer.from(await r.arrayBuffer()), mime: (r.headers.get('content-type') || 'image/png').split(';')[0], name: img.filename };
  }
  const r = await fetch(`${COMFY_URL}/view?filename=${encodeURIComponent(img.filename)}&subfolder=${encodeURIComponent(img.subfolder || '')}&type=output`, { signal: AbortSignal.timeout(60000) }).catch(() => null);
  if (r && r.ok) return { buffer: Buffer.from(await r.arrayBuffer()), mime: (r.headers.get('content-type') || 'image/png').split(';')[0], name: img.filename };
  const full = path.resolve(GENERATED, 'images', img.subfolder || '', img.filename);
  if (!full.startsWith(path.resolve(GENERATED) + path.sep)) throw new Error('bad path');
  return { buffer: await fs.readFile(full), mime: img.filename.endsWith('.png') ? 'image/png' : 'image/jpeg', name: img.filename };
}

// Visible queue: jobs waiting for soucouyant's GPU, in order. Each waiting job carries its
// position (1 = next) so the page can show "2nd in line".
const waiting = [];
function enqueue(job) {
  waiting.push(job);
  job.position = waiting.length;
  notify(job);
}
function dequeue(job) {
  const i = waiting.indexOf(job);
  if (i >= 0) waiting.splice(i, 1);
  job.position = 0;
  waiting.forEach((j, n) => { if (j.position !== n + 1) { j.position = n + 1; notify(j); } });
}

// soucouyant's mover takes finished files out of the generators' output folders within about
// 2 minutes and puts them in kireserver's /media/Generated, mounted here read-only.
const GENERATED = process.env.GENERATED_DIR || '/generated';
async function sendGenerated(res, rel, download) {
  const full = path.resolve(GENERATED, rel);
  if (!full.startsWith(path.resolve(GENERATED) + path.sep)) return res.status(400).json({ error: 'bad path' });
  try { await fs.access(full); } catch { return res.status(404).json({ error: 'file not found yet; try again in a minute' }); }
  res.setHeader('Cache-Control', 'private, max-age=86400');
  return download ? res.download(full) : res.sendFile(full);
}

function trimJobs() {
  while (jobs.size > 500) jobs.delete(jobs.keys().next().value);
}

// KS-03: one library with Kompanion: this user's finished local jobs go to Kompanion once, and the runs they made there show here.

const lastSync = new Map();
async function syncLibrary(req) {
  if (!kompanion.enabled) return;
  const now = Date.now();
  if (now - (lastSync.get(req.userId) || 0) < 15000) return;
  lastSync.set(req.userId, now);

  let changed = false;
  for (const job of jobs.values()) {
    if (job.user !== req.userId || job.imported) continue;
    const payload = importPayload(job, GENERATED);
    if (!payload) continue;
    try {
      await kompanion.importJob(req.userEmail, payload);
      job.imported = true;
      changed = true;
    } catch (e) {
      console.warn(`[KS-03] import ${job.promptId}: ${e.message}`);
    }
  }

  let runs = [];
  let mined = false;
  try {
    runs = await kompanion.mine(req.userEmail);
    mined = true;
  } catch (e) {
    console.warn(`[KS-03] mine: ${e.message}`);
  }

  // STU-D1: a run deleted in Kompanion leaves this library too (/mine lists at most 200 runs,
  // so only a shorter list is complete enough to tell).
  if (mined && runs.length < 200) {
    const live = new Set(runs.map(r => r.id));
    for (const [key, job] of jobs) {
      const runId = kompanionRunId(job);
      if (job.user === req.userId && FINISHED.has(job.status) && runId && !live.has(runId)) {
        jobs.delete(key);
        changed = true;
      }
    }
  }

  const known = new Set();
  for (const job of jobs.values()) {
    known.add(job.promptId);
    if (job.kompanionId) known.add(job.kompanionId);
  }

  for (const run of runs) {
    if (known.has(run.id) || (run.state !== 'done' && run.state !== 'failed')) continue;
    const job = runJob(run, req.userId, req.userEmail);
    jobs.set(job.promptId, job);
    changed = true;
  }

  if (changed) {
    trimJobs();
    saveJobs();
  }
}

// STU-D1: delete finished jobs. A job linked to Kompanion (made there, or imported) deletes its
// Kompanion run too; a local job's files under GENERATED go as well. A running job stays.
const FINISHED = new Set(['done', 'error', 'cancelled']);
function kompanionRunId(job) {
  return job.kompanionId || (job.imported ? job.promptId : null);
}

async function deleteJob(req, id) {
  const job = jobs.get(id);
  if (!job || job.user !== req.userId) return 'missing';
  if (!FINISHED.has(job.status)) return 'running';
  const runId = kompanionRunId(job);
  if (runId && kompanion.enabled) {
    try {
      await kompanion.deleteRun(req.userEmail, runId);
    } catch (e) {
      if (!/404|not found/i.test(e.message)) throw e;
    }
  }
  if (!job.kompanionId) {
    const root = path.resolve(GENERATED);
    const rels = [
      ...(job.images || []).map(i => path.join('images', i.subfolder || '', i.filename)),
      ...(job.files || []).map(f => path.join('audio', f.name))
    ];
    for (const rel of rels) {
      const full = path.resolve(root, rel);
      if (full.startsWith(root + path.sep)) await fs.unlink(full).catch(() => {});
    }
  }
  jobs.delete(id);
  return 'deleted';
}

    res.status(204).end();
  } catch (err) {

});

app.get('/api/events', (req, res) => {
  // Security: Use req.userId only, never query string
  const userId = req.userId;
  if (!userId) {
    return res.status(401).send('Unauthorized');
  }

  res.setHeader('Content-Type', 'text/event-stream');
  res.setHeader('Cache-Control', 'no-cache');
  res.setHeader('Connection', 'keep-alive');

  // Store client reference to remove on close
  const clientSet = new Set();
  let pingInterval = null;

  const sendEvent = (type, data) => {
    const payload = typeof data === 'object' ? JSON.stringify(data) : data;
    res.write(`event: ${type}\ndata: ${payload}\n\n`);
  };

  const sendPing = () => {
    res.write(': ping\n\n');
  };

  // Initial state
  const userJobs = Array.from(jobs.values()).filter(j => j.user === userId);
  userJobs.forEach(job => {
    clientSet.add(job.promptId);
    sendEvent('job', job);
  });

  // Interval for pings
  pingInterval = setInterval(sendPing, 20000);

  // Cleanup on close
  res.on('close', () => {
    clearInterval(pingInterval);
    const set = sseClients.get(userId);
    if (set) { set.delete(res); if (!set.size) sseClients.delete(userId); }
  });

  // Register client
  if (!sseClients.has(userId)) {
    sseClients.set(userId, new Set());
  }
  sseClients.get(userId).add(res);

});

app.get('/api/image', async (req, res) => {
  try {
    const { promptId, filename } = req.query;
    // subfolder may be an empty string, so only require promptId and filename
    
    if (!promptId || !filename) {
      return res.status(400).json({ error: 'Missing parameters' });
    }

    const job = jobs.get(promptId);
    if (!job || job.user !== req.userId) {
      return res.status(403).json({ error: 'Forbidden' });
    }

    // Check if image belongs to this job
    const foundImage = job.images.find(img => img.filename === filename && img.subfolder === (req.query.subfolder || ''));
    if (!foundImage) {
      return res.status(404).json({ error: 'Image not found in job' });
    }

    if (job.kompanionId) return pipeKompanion(res, job, foundImage.kompanion, req.query.download ? foundImage.filename : '');

    // Stream from ComfyUI
    const imageUrl = `${COMFY_URL}/view?filename=${encodeURIComponent(filename)}&subfolder=${encodeURIComponent(req.query.subfolder || '')}&type=output`;
    
    const controller = new AbortController();
    const timeoutId = setTimeout(() => controller.abort(), 60000);

    const response = await fetch(imageUrl, { signal: controller.signal });
    
    if (!response.ok) {
      clearTimeout(timeoutId);
      // Already moved to /media/Generated by soucouyant's mover
      const sub = req.query.subfolder ? `${req.query.subfolder}/` : '';
      return sendGenerated(res, `images/${sub}${filename}`, req.query.download);
    }

    clearTimeout(timeoutId);

    // Copy headers except host
    const contentType = response.headers.get('content-type');
    res.setHeader('Content-Type', contentType);
    res.setHeader('Cache-Control', 'private, max-age=86400');
    
    // Pipe stream using Readable.fromWeb
    stream.Readable.fromWeb(response.body).pipe(res);

  } catch (err) {
    console.error('Image fetch error:', err.message);
    res.status(500).json({ error: err.message });
  }
});
