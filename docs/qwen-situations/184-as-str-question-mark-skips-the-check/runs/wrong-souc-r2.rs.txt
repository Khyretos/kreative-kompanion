//! M6-11: the game planner. A game described in words becomes code tasks (the KG-02 task format) and asset
//! tasks (a library asset to reuse, or a Studio workflow with parameters). The model's answer is checked against
//! the library and the workflows that exist, and mapped onto a KG-02 `Plan` so the same approve-then-apply path
//! writes it into the task tree.
use serde::{Deserialize, Serialize};
use serde_json::{Map, Value, json};

use crate::planner::{self, Issue, Limits, Plan, PlanTask};

/// One asset the game needs: reused from the library (`library_id`) or made by a workflow (`workflow` + `params`).
#[derive(Debug, Clone, Serialize, Deserialize, PartialEq)]
pub struct AssetTask {
    pub key: String,
    pub title: String,
    /// sprite, image, tileset, background, audio, music, sfx, video or model
    pub kind: String,
    #[serde(default)]
    pub tags: Vec<String>,
    #[serde(default)]
    pub library_id: Option<String>,
    #[serde(default)]
    pub workflow: Option<String>,
    #[serde(default)]
    pub params: Map<String, Value>,
    /// keys of other asset tasks that come first
    #[serde(default)]
    pub depends_on: Vec<String>,
    /// keys of the code tasks that need this asset
    #[serde(default)]
    pub needed_by: Vec<String>,
}

#[derive(Debug, Clone, Serialize, Deserialize, PartialEq)]
pub struct GamePlan {
    pub title: String,
    pub goal: String,
    pub code_tasks: Vec<PlanTask>,
    pub asset_tasks: Vec<AssetTask>,
}

/// An asset of the project's library, as far as planning needs it.
#[derive(Debug, Clone)]
pub struct LibAsset {
    pub id: String,
    pub name: String,
    pub kind: String,
    pub tags: Vec<String>,
}

/// One parameter of a Studio workflow. `kind` is "number", "text" or "choice".
#[derive(Debug, Clone)]
pub struct WfParam {
    pub name: String,
    pub kind: String,
    pub choices: Vec<String>,
    pub min: Option<f64>,
    pub max: Option<f64>,
}

/// A Studio workflow as far as planning needs it. `output` is "image", "audio", "video" or "model".
#[derive(Debug, Clone)]
pub struct WfInfo {
    pub name: String,
    pub output: String,
    pub params: Vec<WfParam>,
}

fn issue(kind: &str, key: &str, message: String) -> Issue {
    Issue { kind: kind.into(), key: key.into(), message }
}

/// The JSON schema the model's answer must follow.
pub fn schema() -> Value {
    let task = planner::schema()["properties"]["milestones"]["items"]["properties"]["tasks"]["items"].clone();
    json!({
        "type": "object",
        "title": "Game plan",
        "properties": {
            "goal": { "type": "string" },
            "code_tasks": {
                "type": "array",
                "items": task
            },
            "asset_tasks": {
                "type": "array",
                "items": {
                    "type": "object",
                    "properties": {
                        "key": { "type": "string" },
                        "title": { "type": "string" },
                        "kind": { "type": "string" },
                        "tags": { "type": "array", "items": { "type": "string" } },
                        "library_id": { "type": ["string", "null"] },
                        "workflow": { "type": ["string", "null"] },
                        "params": { "type": "object" },
                        "depends_on": { "type": "array", "items": { "type": "string" } },
                        "needed_by": { "type": "array", "items": { "type": "string" } }
                    },
                    "required": ["key", "title", "kind"]
                }
            }
        },
        "required": ["title", "goal", "code_tasks", "asset_tasks"]
    })
}

/// A game plan from the model's answer.
pub fn parse(text: &str) -> Result<GamePlan, String> {
    let v = crate::taskrun::parse::json_in(text).ok_or("The answer holds no JSON.")?;
    serde_json::from_value(v).map_err(|e| format!("The answer is not a game plan: {e}"))
}

/// The workflow output an asset kind needs.
pub fn output_of_kind(kind: &str) -> Option<&'static str> {
    let k = kind.to_lowercase();
    match k.as_str() {
        "sprite" | "image" | "tileset" | "background" => Some("image"),
        "audio" | "music" | "sfx" => Some("audio"),
        "video" => Some("video"),
        "model" => Some("model"),
        _ => None,
    }
}

/// What is wrong with one parameter value, or None.
pub fn param_issue(p: &WfParam, v: &Value) -> Option<String> {
    match p.kind.as_str() {
        "number" => {
            let num = v.as_f64()?;
            if let Some(m) = p.min {
                if num < m {
                    return Some(format!("Parameter {} must be between {} and {}.", p.name, m, p.max.map(|mx| mx.to_string()).unwrap_or("-".to_string())));
                }
            }
            if let Some(m) = p.max {
                if num > m {
                    return Some(format!("Parameter {} must be between {} and {}.", p.name, p.min.map(|mn| mn.to_string()).unwrap_or("-".to_string()), m));
                }
            }
            None
        }
        "choice" => {
            let s = v.as_str()?;
            if !p.choices.contains(&s.into()) {
                return Some(format!("Parameter {} must be one of {}; \"{}\" is not allowed.", p.name, p.choices.join(", "), s));
            }
            None
        }
        "text" => {
            if v.is_string() {
                None
            } else {
                Some(format!("Parameter {} must be text.", p.name))
            }
        }
        _ => None,
    }
}

/// Everything wrong with the asset tasks: source, library, workflow, parameters, needed_by.
pub fn asset_issues(plan: &GamePlan, _library: &[LibAsset], _workflows: &[WfInfo]) -> Vec<Issue> {
    let mut issues = Vec::new();

    for task in &plan.asset_tasks {
        // Check library_id exists if workflow is None
        if task.workflow.is_none() && task.library_id.is_none() {
            issues.push(issue(
                "missing_source",
                &task.key,
                format!("Asset {} must have a library_id or workflow.", task.title),
            ));
        }

        // Check needed_by references exist in code_tasks
        for key in &task.needed_by {
            if !plan.code_tasks.iter().any(|ct| ct.key == *key) {
                issues.push(issue(
                    "invalid_dependency",
                    &task.key,
                    format!("Asset {} is needed by {}, but no code task has that key.", task.title, key),
                ));
            }
        }

        // Check depends_on references exist in asset_tasks
        for key in &task.depends_on {
            if !plan.asset_tasks.iter().any(|at| at.key == *key) {
                issues.push(issue(
                    "invalid_dependency",
                    &task.key,
                    format!("Asset {} depends on {}, but no asset task has that key.", task.title, key),
                ));
            }
        }
    }

    issues
}

/// Swaps a generated asset for a matching library asset; returns the keys that were swapped.
pub fn prefer_library(plan: &mut GamePlan, _library: &[LibAsset]) -> Vec<String> {
    let mut swapped = Vec::new();

    for task in &mut plan.asset_tasks {
        if task.workflow.is_some() {
            continue;
        }

        if let Some(lib_id) = &task.library_id {
            // Find a matching library asset
            if let Some(lib_asset) = library.iter().find(|la| la.id == *lib_id) {
                // Check if the library asset matches the task's kind and tags
                if lib_asset.kind == task.kind {
                    // Check tag overlap (at least one common tag)
                    let has_common_tag = task.tags.iter().any(|t| lib_asset.tags.contains(t));

                    if has_common_tag {
                        swapped.push(task.key.clone());
                    }
                }
            }
        }
    }

    swapped
}
