(function () {
  'use strict';
  const NS = typeof module === 'object' && module.exports ? null : (globalThis.KKPose = globalThis.KKPose || {});

  // Helper to ensure byte alignment to 4 bytes
  function padToMultiple(arr, multiple) {
    const len = arr.length;
    if (len % multiple === 0) return arr;
    const padded = new Uint8Array((len + multiple - 1) & ~(multiple - 1));
    padded.set(arr);
    // Fill remaining with 0x20 for JSON, 0x00 for BIN
    const fillChar = (arr.length > 0 && arr instanceof Uint8Array && arr[0] !== 0x20) ? 0x20 : 0x00;
    for (let i = arr.length; i < padded.length; i++) {
      padded[i] = fillChar;
    }
    return padded;
  }

  const api = {
    pack(json, bin) {
      const text = JSON.stringify(json);
      const jsonBytes = new TextEncoder().encode(text);
      
      // Pad JSON to multiple of 4
      const paddedJson = padToMultiple(jsonBytes, 4);
      
      // Pad BIN to multiple of 4
      const paddedBin = padToMultiple(bin, 4);
      if (!paddedBin || paddedBin.length === 0) {
        return paddedJson;
      }

      const jsonChunkLen = paddedJson.length;
      const binChunkLen = paddedBin.length;
      const totalLength = 12 + jsonChunkLen + binChunkLen;

      const out = new Uint8Array(totalLength);
      const dv = new DataView(out.buffer, 0, totalLength);

      // Header
      dv.setUint32(0, 0x46546C67, true); // magic "glTF"
      dv.setUint32(4, 2, true);          // version
      dv.setUint32(8, totalLength, true); // total length

      // JSON chunk
      dv.setUint32(12, jsonChunkLen, true); // length
      dv.setUint32(16, 0x4E4F534A, true);   // type "JSON"
      paddedJson.copy(out, 20);

      // BIN chunk (only if bin.length > 0)
      if (bin.length > 0) {
        dv.setUint32(20 + jsonChunkLen, binChunkLen, true); // length
        dv.setUint32(24 + jsonChunkLen, 0x004E4942, true);   // type "BIN"
        paddedBin.copy(out, 28 + jsonChunkLen);
      }

      return out;
    },

    unpack(bytes) {
      if (bytes.length < 20) {
        throw new Error('not a glb file');
      }

      const dv = new DataView(bytes.buffer, bytes.byteOffset, bytes.byteLength);
      const magic = dv.getUint32(0, true);
      if (magic !== 0x46546C67) {
        throw new Error('not a glb file');
      }

      // Read JSON chunk
      const jsonLen = dv.getUint32(12, true);
      const jsonType = dv.getUint32(16, true);
      if (jsonType !== 0x4E4F534A) {
        throw new Error('not a glb file');
      }

      const jsonText = new TextDecoder().decode(bytes.subarray(20, 20 + jsonLen));
      const json = JSON.parse(jsonText);

      let bin = new Uint8Array(0);
      if (jsonLen < bytes.length - 20) {
        const binLen = dv.getUint32(20 + jsonLen, true);
        const binType = dv.getUint32(24 + jsonLen, true);
        if (binType === 0x004E4942) {
          bin = bytes.subarray(28 + jsonLen, 28 + jsonLen + binLen);
        }
      }

      return { json, bin };
    }
  };

  if (NS) NS.glb = api; else module.exports = api;
})();
