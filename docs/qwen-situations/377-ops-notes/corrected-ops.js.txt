(function () {
  'use strict';
  const NS = typeof module === 'object' && module.exports ? null : (globalThis.KKPose = globalThis.KKPose || {});
  const need = (name, file) => (NS ? NS[name] : require(file));
  // STU-PE1: the Coder's draft (3 attempts + 3 repair attempts) lost every note, ignored current values in
  // set_joint and reassigned its parameters; Claude rewrote the op dispatch around one store() (lesson 377).

  const OPS = ['set_joint', 'curl_finger', 'mirror', 'look_at', 'ik', 'move_root', 'reset', 'preset'];
  const KEYS = ['bend', 'side', 'twist'];
  const round1 = (x) => Math.round(x * 10) / 10;
  const isNum = (x) => typeof x === 'number' && Number.isFinite(x);

  // Each value (missing = 0) clamped to the joint's limits; -0 becomes 0.
  function clamp(name, e) {
    const lim = need('skeleton', './skeleton.js').byName[name].limits;
    const out = {};
    for (const k of KEYS) out[k] = Math.min(lim[k][1], Math.max(lim[k][0], (e && e[k]) || 0)) || 0;
    return out;
  }

  // Every stored joint value goes through here, so the notes say what the limits changed.
  function store(p, name, e, notes) {
    const c = clamp(name, e);
    for (const k of KEYS) {
      const want = (e && e[k]) || 0;
      if (Math.abs(want - c[k]) > 1e-9) notes.push(`${name} ${k} ${round1(want)} limited to ${c[k]}`);
    }
    p.joints[name] = c;
  }

  function setJoint(p, op, notes) {
    const S = need('skeleton', './skeleton.js');
    if (!S.byName[op.joint]) throw new Error(`Unknown joint "${op.joint}".`);
    const cur = p.joints[op.joint] || {};
    const e = {};
    for (const k of KEYS) {
      e[k] = cur[k] || 0;
      if (op[k] === undefined) continue;
      if (!isNum(op[k])) throw new Error(`${k} must be a number.`);
      e[k] = op.add ? e[k] + op[k] : op[k];
    }
    store(p, op.joint, e, notes);
  }

  function curlFinger(p, op, notes) {
    const S = need('skeleton', './skeleton.js');
    if (!['l', 'r', 'both'].includes(op.hand)) throw new Error('hand must be l, r or both.');
    const fingers = Object.keys(S.FINGERS);
    if (op.finger !== 'all' && !fingers.includes(op.finger)) throw new Error('finger must be thumb, index, middle, ring, pinky or all.');
    if (!isNum(op.amount) || op.amount < 0 || op.amount > 1) throw new Error('amount must be between 0 and 1.');
    for (const s of op.hand === 'both' ? ['l', 'r'] : [op.hand]) {
      for (const f of op.finger === 'all' ? fingers : [op.finger]) {
        for (const j of S.FINGERS[f]) {
          const name = `${s}_${j}`;
          const cur = p.joints[name] || {};
          store(p, name, { bend: op.amount * S.byName[name].limits.bend[1], side: cur.side, twist: cur.twist }, notes);
        }
      }
    }
  }

  function mirror(p, op, notes) {
    const S = need('skeleton', './skeleton.js');
    if (op.from === 'l' || op.from === 'r') {
      const other = op.from === 'l' ? 'r_' : 'l_';
      for (const name of Object.keys(p.joints)) if (name.startsWith(other)) delete p.joints[name];
      for (const name of Object.keys(p.joints)) {
        if (name.startsWith(op.from + '_')) store(p, S.mirrorName(name), { ...p.joints[name] }, notes);
      }
    } else if (op.from === 'swap') {
      const joints = {};
      for (const [name, v] of Object.entries(p.joints)) {
        if (name.startsWith('l_') || name.startsWith('r_')) joints[S.mirrorName(name)] = v;
        else joints[name] = { bend: v.bend, side: 0 - v.side || 0, twist: 0 - v.twist || 0 };
      }
      p.joints = joints;
      p.root.pos[0] = 0 - p.root.pos[0] || 0;
    } else {
      throw new Error('from must be l, r or swap.');
    }
  }

  function applyOne(p, op, notes) {
    const FK = need('fk', './fk.js');
    switch (op && op.op) {
      case 'set_joint': return setJoint(p, op, notes);
      case 'curl_finger': return curlFinger(p, op, notes);
      case 'mirror': return mirror(p, op, notes);
      case 'look_at': {
        const r = need('lookat', './lookat.js').lookAt(p, op);
        for (const [name, e] of Object.entries(r.joints)) store(p, name, e, notes);
        return p;
      }
      case 'ik': {
        const IK = need('ik', './ik.js');
        if (!IK.CHAINS[op.chain]) throw new Error('chain must be l_arm, r_arm, l_leg or r_leg.');
        const r = IK.solve(p, op);
        for (const [name, e] of Object.entries(r.joints)) store(p, name, e, notes);
        if (!r.reached) notes.push(`${op.chain} cannot reach the target`);
        return p;
      }
      case 'move_root':
        if (!Array.isArray(op.pos) || op.pos.length !== 3 || !op.pos.every(isNum)) throw new Error('pos must be [x, y, z].');
        p.root.pos = op.pos.slice();
        return p;
      case 'reset':
        if (!op.joints) return FK.defaultPose();
        for (const name of op.joints) delete p.joints[name];
        return p;
      case 'preset': {
        const PR = need('presets', './presets.js');
        if (!PR.PRESETS[op.name]) throw new Error(`Unknown preset "${op.name}". Use one of: ${Object.keys(PR.PRESETS).join(', ')}.`);
        const r = apply(FK.defaultPose(), PR.PRESETS[op.name]);
        notes.push(...r.notes);
        return r.pose;
      }
      default:
        throw new Error(`Unknown op "${op && op.op}". Use one of: ${OPS.join(', ')}.`);
    }
  }

  // ops: one op or a list. Never changes the input pose. Returns {pose, notes}.
  function apply(pose, ops) {
    let p = JSON.parse(JSON.stringify(pose));
    const notes = [];
    for (const op of Array.isArray(ops) ? ops : [ops]) p = applyOne(p, op, notes) || p;
    for (const [name, v] of Object.entries(p.joints)) if (!v.bend && !v.side && !v.twist) delete p.joints[name];
    return { pose: p, notes };
  }

  const api = { OPS, clamp, apply };
  if (NS) NS.ops = api; else module.exports = api;
})();
