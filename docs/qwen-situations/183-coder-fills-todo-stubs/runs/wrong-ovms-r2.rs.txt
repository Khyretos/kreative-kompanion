//! M6-11: the game planner. A game described in words becomes code tasks (the KG-02 task format) and asset
//! tasks (a library asset to reuse, or a Studio workflow with parameters). The model's answer is checked against
//! the library and the workflows that exist, and mapped onto a KG-02 `Plan` so the same approve-then-apply path
//! writes it into the task tree.
use serde::{Deserialize, Serialize};
use serde_json::{Map, Value, json};

use crate::planner::{self, Issue, Limits, Milestone, Plan, PlanTask};

/// One asset the game needs: reused from the library (`library_id`) or made by a workflow (`workflow` + `params`).
#[derive(Debug, Clone, Serialize, Deserialize, PartialEq)]
pub struct AssetTask {
    pub key: String,
    pub title: String,
    /// sprite, image, tileset, background, audio, music, sfx, video or model
    pub kind: String,
    #[serde(default)]
    pub tags: Vec<String>,
    #[serde(default)]
    pub library_id: Option<String>,
    #[serde(default)]
    pub workflow: Option<String>,
    #[serde(default)]
    pub params: Map<String, Value>,
    /// keys of other asset tasks that come first
    #[serde(default)]
    pub depends_on: Vec<String>,
    /// keys of the code tasks that need this asset
    #[serde(default)]
    pub needed_by: Vec<String>,
}

#[derive(Debug, Clone, Serialize, Deserialize, PartialEq)]
pub struct GamePlan {
    pub title: String,
    pub goal: String,
    pub code_tasks: Vec<PlanTask>,
    pub asset_tasks: Vec<AssetTask>,
}

/// An asset of the project's library, as far as planning needs it.
#[derive(Debug, Clone)]
pub struct LibAsset {
    pub id: String,
    pub name: String,
    pub kind: String,
    pub tags: Vec<String>,
}

/// One parameter of a Studio workflow. `kind` is "number", "text" or "choice".
#[derive(Debug, Clone)]
pub struct WfParam {
    pub name: String,
    pub kind: String,
    pub choices: Vec<String>,
    pub min: Option<f64>,
    pub max: Option<f64>,
}

/// A Studio workflow as far as planning needs it. `output` is "image", "audio", "video" or "model".
#[derive(Debug, Clone)]
pub struct WfInfo {
    pub name: String,
    pub output: String,
    pub params: Vec<WfParam>,
}

fn issue(kind: &str, key: &str, message: String) -> Issue {
    Issue { kind: kind.into(), key: key.into(), message }
}

/// The JSON schema the model's answer must follow.
pub fn schema() -> Value {
    let asset = json!({
        "type": "object",
        "properties": {
            "key": { "type": "string", "description": "short unique id, used in depends_on" },
            "title": { "type": "string" },
            "kind": { "type": "string", "description": "sprite, image, tileset, background, audio, music, sfx, video or model" },
            "tags": { "type": "array", "items": { "type": "string" } },
            "library_id": { "type": ["string", "null"], "description": "id of a library asset to reuse" },
            "workflow": { "type": ["string", "null"], "description": "name of a Studio workflow" },
            "params": { "type": "object", "description": "parameters for the workflow" },
            "depends_on": { "type": "array", "items": { "type": "string" }, "description": "keys of other asset tasks that come first" },
            "needed_by": { "type": "array", "items": { "type": "string" }, "description": "keys of the code tasks that need this asset" }
        },
        "required": ["key", "title", "kind"]
    });
    json!({
        "type": "object",
        "properties": {
            "title": { "type": "string" },
            "goal": { "type": "string" },
            "code_tasks": {
                "type": "array",
                "items": {
                    "type": "object",
                    "properties": {
                        "key": { "type": "string" },
                        "title": { "type": "string" },
                        "goal": { "type": "string" },
                        "steps": { "type": "array", "items": { "type": "string" } },
                        "done_when": { "type": "string" },
                        "files": { "type": "array", "items": { "type": "string" } },
                        "lines": { "type": "integer" },
                        "depends_on": { "type": "array", "items": { "type": "string" } }
                    },
                    "required": ["key", "title", "goal", "steps", "done_when", "files", "lines", "depends_on"]
                }
            },
            "asset_tasks": {
                "type": "array",
                "items": asset
            }
        },
        "required": ["title", "goal", "code_tasks", "asset_tasks"]
    })
}

/// A game plan from the model's answer.
pub fn parse(text: &str) -> Result<GamePlan, String> {
    let v = crate::taskrun::parse::json_in(text).ok_or("The answer holds no JSON.")?;
    serde_json::from_value(v).map_err(|e| format!("The answer is not a game plan: {e}"))
}

/// The workflow output an asset kind needs.
pub fn output_of_kind(kind: &str) -> Option<&'static str> {
    let k = kind.to_ascii_lowercase();
    match k.as_str() {
        "sprite" | "image" | "tileset" | "background" => Some("image"),
        "audio" | "music" | "sfx" => Some("audio"),
        "video" => Some("video"),
        "model" => Some("model"),
        _ => None,
    }
}

/// What is wrong with one parameter value, or None.
pub fn param_issue(p: &WfParam, v: &Value) -> Option<String> {
    if p.kind == "number" {
        let num = v.as_f64()?;
        if let (Some(min), Some(max)) = (p.min, p.max) {
            if num < min || num > max {
                return Some(format!("{} must be between {} and {}", p.name, min, max));
            }
        } else if p.min.is_some() && num < *p.min.unwrap() {
            return Some(format!("{} must be at least {}", p.name, p.min.unwrap()));
        } else if p.max.is_some() && num > *p.max.unwrap() {
            return Some(format!("{} must be at most {}", p.name, p.max.unwrap()));
        }
        return None;
    }
    if p.kind == "choice" {
        if !p.choices.contains(&v.as_str()?.to_string()) {
            return Some(format!("{} must be one of {:?}", p.name, p.choices));
        }
        return None;
    }
    // text: any non-empty string is OK
    if v.is_null() || v.as_str().is_none() || v.as_str().unwrap().is_empty() {
        return Some(format!("{} must be a non-empty string", p.name));
    }
    None
}

/// Everything wrong with the asset tasks: source, library, workflow, parameters, needed_by.
pub fn asset_issues(plan: &GamePlan, library: &[LibAsset], workflows: &[WfInfo]) -> Vec<Issue> {
    let mut out = Vec::new();
    let known_lib_ids: HashSet<_> = library.iter().map(|a| a.id.as_str()).collect();
    let known_wf_names: HashSet<_> = workflows.iter().map(|w| w.name.as_str()).collect();
    let kind_to_output: HashMap<_, _> = library.iter().chain(workflows.iter()).flat_map(|a| {
        if let Some(out) = output_of_kind(&a.kind) {
            vec![(&a.kind, out)]
        } else {
            vec![]
        }
    }).collect();
    let known_outputs: HashSet<_> = kind_to_output.values().copied().collect();
    let wf_outputs: HashMap<_, _> = workflows.iter().map(|w| (&w.name, &w.output)).collect();

    for a in &plan.asset_tasks {
        // exactly one source
        let has_lib = a.library_id.is_some();
        let has_wf = a.workflow.is_some();
        if !has_lib && !has_wf {
            out.push(issue("no_source", &a.key, format!("Asset \"{}\" has neither a library asset nor a workflow.", a.title)));
        } else if has_lib && has_wf {
            out.push(issue("two_sources", &a.key, format!("Asset \"{}\" has both a library asset and a workflow.", a.title)));
        }

        // library asset must exist and match the kind
        if let Some(lib_id) = &a.library_id {
            if !known_lib_ids.contains(lib_id.as_str()) {
                out.push(issue("unknown_asset", &a.key, format!("Library asset \"{}\" does not exist.", lib_id)));
            } else {
                let found = library.iter().find(|l| l.id == *lib_id);
                if let Some(l) = found {
                    if l.kind != a.kind {
                        out.push(issue("wrong_kind", &a.key, format!("Library asset \"{}\" is a {} but the asset task is a {}.", lib_id, l.kind, a.kind)));
                    }
                }
            }
        }

        // workflow must exist and produce the right output
        if let Some(wf_name) = &a.workflow {
            if !known_wf_names.contains(wf_name.as_str()) {
                out.push(issue("unknown_workflow", &a.key, format!("Workflow \"{}\" does not exist.", wf_name)));
            } else {
                let wf = workflows.iter().find(|w| w.name == *wf_name).unwrap();
                let expected_out = wf_outputs.get(wf_name).unwrap();
                let actual_out = output_of_kind(&a.kind);
                if actual_out != Some(expected_out) {
                    out.push(issue("wrong_output", &a.key, format!("Workflow \"{}\" produces {} but the asset is a {}.", wf_name, expected_out, a.kind)));
                }
            }
        }

        // check parameters against the workflow
        if let Some(wf_name) = &a.workflow {
            let wf = workflows.iter().find(|w| w.name == *wf_name).unwrap();
            for (k, v) in &a.params {
                if let Some(param) = wf.params.iter().find(|p| p.name == *k) {
                    if let Some(msg) = param_issue(param, v) {
                        out.push(issue("bad_param", &a.key, format!("Parameter {} of workflow \"{}\": {}", k, wf_name, msg)));
                    }
                } else {
                    out.push(issue("unknown_param", &a.key, format!("Parameter \"{}\" is not defined in workflow \"{}\".", k, wf_name)));
                }
            }
        }

        // unknown kind
        if output_of_kind(&a.kind).is_none() {
            out.push(issue("unknown_kind", &a.key, format!("Asset kind \"{}\" is not supported.", a.kind)));
        }
    }

    out
}

/// Swaps a generated asset for a matching library asset; returns the keys that were swapped.
pub fn prefer_library(plan: &mut GamePlan, library: &[LibAsset]) -> Vec<String> {
    let mut swapped = Vec::new();
    let known_lib_ids: HashSet<_> = library.iter().map(|a| a.id.as_str()).collect();
    let known_lib_kinds: HashMap<_, _> = library.iter().map(|a| (&a.kind, &a.tags)).collect();

    for a in &mut plan.asset_tasks {
        if a.library_id.is_some() {
            continue;
        }
        if a.workflow.is_some() {
            continue;
        }
        // find a library asset with the same kind and best tag match
        let mut best: Option<&LibAsset> = None;
        for l in library {
            if l.kind != a.kind {
                continue;
            }
            // count matching tags
            let score = a.tags.iter().filter(|t| l.tags.contains(t)).count();
            if best.is_none() || score > best.as_ref().unwrap().tags.iter().filter(|t| a.tags.contains(t)).count() {
                best = Some(l);
            }
        }
        if let Some(l) = best {
            a.library_id = Some(l.id.clone());
            swapped.push(a.key.clone());
        }
    }
    swapped
}

/// The game plan as a KG-02 plan: an "Assets" milestone, then a "Code" milestone.
pub fn to_plan(plan: &GamePlan) -> Plan {
    let mut milestones = Vec::new();

    // Assets milestone
    if !plan.asset_tasks.is_empty() {
        let mut tasks = Vec::new();
        for a in &plan.asset_tasks {
            let mut goal = format!("Create the {} asset \"{}\".", a.kind, a.title);
            if let Some(lib_id) = &a.library_id {
                goal.push_str(&format!(" Use library asset \"{}\".", lib_id));
            } else if let Some(wf_name) = &a.workflow {
                goal.push_str(&format!(" Generate it with workflow \"{}\".", wf_name));
            }
            if !a.params.is_empty() {
                let params: Vec<_> = a.params.iter().map(|(k, v)| format!("{}={:?}", k, v)).collect();
                goal.push_str(&format!(" Params: {}", params.join(", ")));
            }
            tasks.push(PlanTask {
                key: a.key.clone(),
                title: a.title.clone(),
                goal,
                steps: vec![],
                done_when: "The asset is ready and saved.".to_string(),
                files: vec![],
                lines: 0,
                depends_on: a.depends_on.clone(),
            });
        }
        milestones.push(Milestone {
            title: "Assets".into(),
            goal: "All assets are ready.".into(),
            tasks,
        });
    }

    // Code milestone
    if !plan.code_tasks.is_empty() {
        let mut tasks = Vec::new();
        for t in &plan.code_tasks {
            let mut goal = t.goal.clone();
            if !t.files.is_empty() {
                goal.push_str(&format!(" Edit {} file(s).", t.files.len()));
            }
            tasks.push(t.clone());
        }
        milestones.push(Milestone {
            title: "Code".into(),
            goal: "All code tasks are done.".into(),
            tasks,
        });
    }

    Plan {
        title: plan.title.clone(),
        goal: plan.goal.clone(),
        milestones,
    }
}

/// All rules: the KG-02 task rules on the mapped plan plus the asset rules.
pub fn all_issues(plan: &GamePlan, lim: &Limits, library: &[LibAsset], workflows: &[WfInfo]) -> Vec<Issue> {
    let mut out = asset_issues(plan, library, workflows);
    let p = to_plan(plan);
    out.extend(planner::issues(&p, lim));
    out
}

/// Asks the model for a game plan, up to 3 rounds while the answer breaks a rule.
pub async fn generate<F, Fut>(mut ask: F, system: &str, user: &str, lim: &Limits, library: &[LibAsset], workflows: &[WfInfo]) -> Result<(GamePlan, Vec<Issue>), String>
where
    F: FnMut(Vec<Value>) -> Fut,
    Fut: std::future::Future<Output = Result<String, String>>,
{
    let mut messages = vec![json!({ "role": "system", "content": system }), json!({ "role": "user", "content": user })];
    let mut last: Option<(GamePlan, Vec<Issue>)> = None;
    let mut last_err = String::from("The model made no game plan.");
    for _ in 0..3 {
        let text = ask(messages.clone()).await?;
        messages.push(json!({ "role": "assistant", "content": text }));
        match parse(&text) {
            Ok(plan) => {
                let found = all_issues(&plan, lim, library, workflows);
                // swap library assets before checking
                prefer_library(&mut plan, library);
                let found_after = all_issues(&plan, lim, library, workflows);
                if found_after.is_empty() {
                    return Ok((plan, found_after));
                }
                messages.push(json!({ "role": "user", "content": format!("The plan breaks these rules:\n{}", found_after.iter().map(|i| format!("- {} ({}): {}", i.key, i.kind, i.message)).collect::<Vec<_>>().join("\n")) }));
                last = Some((plan, found_after));
            }
            Err(e) => {
                messages.push(json!({ "role": "user", "content": format!("{e} Answer only with the JSON game plan.") }));
                last_err = e;
            }
        }
    }
    last.ok_or(last_err)
}
