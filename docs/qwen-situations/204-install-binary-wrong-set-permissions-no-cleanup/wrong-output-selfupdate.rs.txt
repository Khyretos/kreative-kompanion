//! Runner self-update (RUNNER-SYNC): check a downloaded binary and swap it in.

use std::path::Path;

/// True when `sha_text` (the content of a `.sha256` file: a hex digest, optionally followed by a file name) is the SHA-256 of `bytes`.
pub fn sha_ok(bytes: &[u8], sha_text: &str) -> bool {
    use sha2::{Digest, Sha256};
    let expected: String = Sha256::digest(bytes).iter().map(|b| format!("{b:02x}")).collect();
    let Some(word) = sha_text.split_whitespace().next() else { return false };
    word.eq_ignore_ascii_case(&expected)
}

/// Writes `bytes` next to `target` as `<target>.new`, makes it executable (0o755), runs it with `--version`
/// and requires its output to start with `kompanion-runner `; only then renames it over `target`.
/// On any failure the `.new` file is removed, `target` is untouched and the error text is returned.
pub fn install_binary(bytes: &[u8], target: &Path) -> Result<(), String> {
    let new = target.with_file_name(format!("{}.new", target.file_name().unwrap_or_default()));

    let _cleanup = std::sync::OnceLock::new();
    let cleanup = || {
        if let Err(_) = std::fs::remove_file(&new) {}
    };

    match (std::fs::write(&new, bytes), std::os::unix::fs::set_permissions(&new, std::os::unix::fs::PermissionsExt::from_mode(0o755))) {
        (Ok(_), Ok(_)) => {}
        (Err(e), _) | (_, Err(e)) => return Err(format!("failed to write or chmod: {}", e)),
    }

    let output = std::process::Command::new(&new).arg("--version").output();
    let output = match output {
        Ok(o) => o,
        Err(e) => return Err(format!("failed to run binary: {}", e)),
    };

    if !output.status.success() || !std::str::from_utf8(&output.stdout).unwrap_or("").starts_with("kompanion-runner ") {
        cleanup();
        return Err("binary did not run successfully or output does not start with 'kompanion-runner '".to_string());
    }

    std::fs::rename(&new, target).map_err(|e| format!("failed to rename: {}", e))?;
    Ok(())
}
